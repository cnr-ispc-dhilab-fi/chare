Evaluation of Echoes of Loss - Demographics and Single Modules

We provide an analysis of the data collected during the evaluation of the caring prototype Echoes of Loss at Digital Heritage 2025 and ArcheoVirtual Expos. The experimental design foresees two different test filled before and after the experience (repeated measures).

There are therefore two different datasets:

  • pre-test: 90 participants
  • post-test: 70 participants

All participants have been pseudo-anonymised: we upload the dataset as dataframes and focus only on the visitors who take parts in both the evaluation (n=70)

get_emoval <- function(df_emotions) {
  round(mean(as.numeric(df_emotions), na.rm = TRUE) / 5, 3)
}

plot_plutchik <- function(emotions) {
  # emotions deve essere una lista o un named numeric vector
  emo_df <- data.frame(
    emotion = names(emotions),
    value = as.numeric(unlist(emotions))
  )
  
  # Ordino in modo circolare
  emo_df$emotion <- factor(emo_df$emotion, levels = emo_df$emotion)
  
  ggplot(emo_df, aes(x = emotion, y = value, fill = emotion)) +
    geom_bar(stat = "identity", width = 1, color = "white") +
    coord_polar(start = 0) +
    scale_fill_manual(values = c(
      "joy"          = "#FFD700",  # giallo
      "trust"        = "#00BFAE",  # verde acqua
      "fear"         = "#1E90FF",  # blu
      "surprise"     = "#9370DB",  # viola
      "sadness"      = "#4682B4",  # blu-grigio
      "disgust"      = "#228B22",  # verde scuro
      "anger"        = "#DC143C",  # rosso
      "anticipation" = "#FFA500"   # arancione
    )) +
    theme_minimal(base_size = 14) +
    theme(
      legend.position = "none",
      axis.title = element_blank(),
      axis.text.y = element_blank(),
      axis.ticks = element_blank(),
      panel.grid = element_blank()
    ) +
    ggtitle("Plutchik-like emotions (scaled 0–1)")
}

classify_response <- function(val) {
  
  # Missing or empty
  if (is.na(val) || val == "") return("C")
  
  steps <- unlist(strsplit(val, ";"))
  steps <- steps[steps != ""]
  steps_low <- tolower(steps)
  
  # Must start with historic and end with mock-ups
  if (!str_detect(steps_low[1], "historic") ||
      !str_detect(steps_low[length(steps_low)], "mock")) {
    return("C")
  }
  
  # Middle steps
  middle <- steps_low[-c(1, length(steps_low))]
  
  required <- c("microscope", "uvl", "xrf")
  
  # A: contains all three in any order
  if (all(map_lgl(required, ~ any(str_detect(middle, .x))))) {
    
    # B: correct order microscope → uvl → xrf
    pos <- sapply(required, function(x)
      which(str_detect(middle, x))[1]
    )
    
    if (all(diff(pos) > 0)) {
      return("B")
    } else {
      return("A")
    }
  }
  
  return("C")
}

# Apply classification
complete_df$class <- sapply(
  complete_df$`Could you order the process of the analysis for colour reconstruction as presented in this exhibition?`,
  classify_response
)
